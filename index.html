Part I — Personal Note (November 2026)

How It Feels Rediscovering This Code 13 Years Later

When you spend 15+ years in software engineering—moving from India to Amsterdam, and building systems across ThoughtWorks, Freshworks, Booking.com, and as a Staff Software Engineer at ServiceNow—it is fun to look back at where your engineering instincts started.

Finding Machine Coding Round - Raj Sekharan(1).eml in my archives felt like opening a sealed letter from my 24-year-old self. Look at the context of that Wednesday night on February 20, 2013: there was no AI, no ES6, no CSS Grid or Flexbox, and Flipkart’s prompt arrived at 5:14 PM after a full workday. It banned jQuery at a time when 95% of web developers couldn’t attach a click listener without $(), and demanded compatibility back to Internet Explorer 7.

What moves me most looking at spreadsheet.js today isn’t that it was finished—it wasn’t. What moves me is how my 24-year-old brain chose to spend those five hours. Instead of hacking together a cheap 3×3 HTML <table> with ugly buttons just to tick boxes, I opened Notepad++ (leaving behind the telltale new 3.txt scratch file) and tried to build a real desktop spreadsheet instrument: a closure-encapsulated LayoutManager, a top-left #junction origin block, Smalltalk/Ruby-style columnDo and rowDo higher-order iterators using callback.apply(el), my own cross-browser bind() shim, coordinate-tagged cells (columnoffsetval) that survive arbitrary row and column deletions without shifting-index bugs, and a custom right-click context menu.

At 10:32 PM, exhausted and out of time before wiring up column sorting, I didn’t make up excuses or hide. I zipped the folder, wrote two honest sentences—“Attached is my solution. It’s partial and doesn’t do everything you asked. But here is anyway FWIW.”—and hit send. And I passed the coding round—Flipkart called me in for the interview anyway, because the engineer who read the code saw immediately that it was 100% original, hand-crafted vanilla JavaScript architecture. (While I passed the coding round, the role ultimately did not pan out for other reasons.)

Looking back at this 3.5 KB zip file 13 years later, it is fun to see how much browser engineering has changed since the IE7 days—and how the habit of treating a web page like a desktop UI layout manager (LayoutManager) was already there.

02 · ThoughtWorks Ruby & Smalltalk DNA

Higher-Order Iterators (columnDo & rowDo)

Named after Smalltalk/Ruby do blocks from ThoughtWorks, columnDo and rowDo pass each matching DOM cell as this via callback.apply(el)—rebuilding jQuery’s .each() mechanism from scratch and reusing it across select, unselect, and delete. 

columnDo(selectedColumn, function() {
    if (this === null || this === undefined)
        return;
    this.parentNode.removeChild(this);
});
03 · Coordinate Invariants

Solving the “Shifting Index on Delete” Bug

If you delete Column 2 using raw DOM child indices (childNodes[2]), Column 3 shifts to index 2 and subsequent deletions corrupt the grid. By stamping every cell with logical coordinate attributes (columnoffsetval and row), deleting arbitrary rows or columns never broke subsequent selections. 

var offset = parseInt(el.getAttribute("columnoffsetval"));
if (offset == index) {
    callback.apply(el);
}
04 · Zero-Library Browser Shims

Hand-Rolled Cross-Browser bind() & IE7 Shims

Because jQuery was prohibited and IE7 was required, I wrote custom shims for event binding (attachEvent vs. addEventListener), target resolution (e.srcElement vs. e.target), and even compensated for W3C whitespace text-node offsets in childNodes. 

var bind = function (ele, e, callback) {
    if (ele.attachEvent) {
        ele.attachEvent("onclick", callback);
    } else {
        ele.addEventListener(e, callback);
    }
};
05 · Desktop GUI Mindset

LayoutManager, #junction & Context Menus

Borrowing LayoutManager from Java Swing / Android UI architecture, I built a full-viewport div grid with a 30×30px top-left #junction origin block, row/column selector bars, and context-sensitive right-click menus (#menu-row vs. #menu-col). 

if (document.attachEvent) {
    document.attachEvent("oncontextmenu", handler);
} else {
    document.addEventListener("contextmenu", handler);
}
06 · Honest Retrospective Lesson

Why the 5-Hour Clock Ran Out (DOM vs. 2D State)

Why did I run out of time before finishing Sort and Add Row? Because I spent 3.5 hours perfecting the desktop-style viewport chrome and mutated 2,500 DOM nodes directly instead of backing the grid with a tiny 2D array (grid[r][c]). It was the classic craftsman trade-off—and the exact reason the reviewer still said yes.

    var LayoutManager = function () {
		var defaultRowCount = 50;
		var defaultColumnCount = 50;
		
		var currentRowCount = defaultRowCount;
		var currentColumnCount = defaultColumnCount;
		
		var canvas = document.getElementById("container");
		
		var headerContainer = document.getElementById("header");
		var bodyContainer = document.getElementById("body");
	
		var getAvailableWidth = function () {
			var width;
			if (self.innerHeight) {
					width = self.innerWidth;
			}
			else if (document.documentElement && document.documentElement.clientHeight) {
					width = document.documentElement.clientWidth;
			}
			else if (document.body) {
					width = document.body.clientWidth;
			}
			return width;
		};
		
		var getAvaialbleHeight = function () {
		
			var height = 0;
			if (self.innerHeight) {
					height = self.innerHeight;
			}
			else if (document.documentElement && document.documentElement.clientHeight) {
					height = document.documentElement.clientHeight;
			}
			else if (document.body) {
					height = document.body.clientHeight;
			}
			return height;
		};
		
		var selectedColumn = null;
		var selectedRow = null;
		
		var unselectColumn = function (index) {
		
		    columnDo(index, function() {
				if (this === undefined)
					return;
				var sclassName = this.className;
				sclassName = sclassName.replace("selected","");
				this.className = sclassName;
			});
			
		};
		
		var unselectRow = function (index) {
		
		    rowDo(index, function() {
				if (this === undefined)
					return;
				var sclassName = this.className;
				sclassName = sclassName.replace("selected","");
				this.className = sclassName;
			});
			
		};
		
		var columnDo = function (index, callback) {
			
			for (var iter=0; iter < currentRowCount; iter++) {
				//get the appropriate row
				var row = bodyContainer.childNodes[iter];
				
				for (var subiter = 1; subiter < row.childNodes.length; subiter++) {
					var el = row.childNodes[subiter];
					var offset = parseInt(el.getAttribute("columnoffsetval"));
					if (offset == index) {
				       callback.apply(el);
					}
				}
				
			}
		}
		
		var rowDo = function (index, callback) {
			
			for (var iter=1; iter <=currentColumnCount; iter++) {
				//get the appropriate row
				var i = index;
				if (!window.attachEvent)
					i = index+1;
				var cell = bodyContainer.childNodes[i].childNodes[iter];
				callback.apply(cell);
			}
		};
		
		
		var selectColumn = function (columnIndex) {
		
			if (columnIndex === null || columnIndex === undefined) {
				return;
			}
			
			resetSelection();
			selectedColumn = columnIndex;
			columnDo(columnIndex, function () {
				if (this === undefined)
					return;
				var sClass = this.className;
				if (sClass.indexOf("selected") != -1 )
					return;
				else {
					sClass += " selected";
					this.className = sClass;
				}

			});
		
		};
		
		var resetSelection = function () {
		    if (selectedColumn !== null || selectedRow !== null) {
				unselectRow(selectedRow);
				unselectColumn(selectedColumn);
				selectedColumn = null;
				selectedRow = null;
			}
		};
		
		
		var selectRow = function (rowIndex) {
		
			if (rowIndex === null || rowIndex === undefined) {
				return;
			}
			resetSelection();
			selectedRow = rowIndex;
			rowDo(rowIndex, function () {
				var sClass = this.className;
				if (sClass.indexOf("selected") != -1 )
					return;
				else {
					sClass += " selected";
					this.className = sClass;
				}

			});
		
		};
		
		var bind = function (ele, e, callback) {
			if (ele.attachEvent) {
				ele.attachEvent("onclick", callback);
			}
			else {
				ele.addEventListener(e, callback);
			}
		};
		
		
		
		var init = function () {
			
			
			var createHeaderCell = function () {
				var headerCell = document.createElement("div");
				headerCell.setAttribute("class", "table-head");
				headerCell.setAttribute("style", "background-color: #ccc; border-bottom: 1px solid #000");
				
				return headerCell;
			};
			
			for (var iter = 0; iter < defaultColumnCount; iter++ ) {
				var headerElement = createHeaderCell();
				headerElement.setAttribute("id","col-"+iter);
				headerElement.setAttribute("columnoffsetval", iter);
				bind(headerElement, "click",  function(e) {
					var el = (e.srcElement)?e.srcElement:e.target;
					selectColumn(parseInt(el.getAttribute("columnoffsetval")));
				});
				
				headerElement.innerHTML= iter;
				
				headerContainer.appendChild(headerElement);
			}
			
			
			var createRowSelector = function () {
				var selector = document.createElement("div");
				selector.setAttribute("style","height: 30px; width: 30px; border-right: 1px solid #cdc; border-top: 1px solid #aaa; border-bottom: 1px solid #eee");
				selector.setAttribute("class", "table-row-selector");

				return selector;
			}
			
			var deleteColumn = function () {
				//remove table header
				if (selectedColumn === null)
					return
				var headercell = document.getElementById("col-"+selectedColumn);
				headercell.parentNode.removeChild(headercell);
				//remove all rows for that column
				
				columnDo(selectedColumn, function() {
					if (this === null || this === undefined)
						return;
					this.parentNode.removeChild(this);
				});
				
				dismisContext();
			
			};
			
			
			var delCol = document.getElementById("deletecol");
			
			bind(delCol, "click", deleteColumn); 
			
			var deleteRow = function() {
				if (selectedRow === null)
					return;
				var tableBody = document.getElementById("body");
				
				for (var iter=1; iter < tableBody.childNodes.length; iter++) {

					var current = tableBody.childNodes[iter];
					if (!current)
						continue;
					if (selectedRow == parseInt(current.getAttribute("row")))
						current.parentNode.removeChild(current);
				}
				
				selectedRow = null ;
					
				dismisContext();
			
			};
			
			var delrowmenu = document.getElementById("deleterow");
			
			bind(delrowmenu, "click", deleteRow);
			
			var getDefaultCell = function () {
				var cell = document.createElement("div");
				cell.setAttribute("class","table-cell");
				bind(cell, "click", function(e) {
					el  = (e.srcElement)?e.srcElement:e.target;
					el.setAttribute("contentEditable","true");
				});
				
				var contentEditableDisble = function (e) {
					el  = (e.srcElement)?e.srcElement:e.target;
					el.setAttribute("contentEditable","true");
				};
				
				if (cell.attachEvent) {
					cell.attachEvent("onblur", contentEditableDisble);
				}
				return cell;
			};
			
			var getRow = function () {
			    var row = document.createElement("div");
				row.setAttribute("class","table-row");
				return row;
			};
			
			for (var rowIter=0; rowIter< defaultRowCount; rowIter++) {
			    var row = getRow();
				row.setAttribute("row",rowIter);
				bodyContainer.appendChild(row);

				rowSelector.setAttribute("id","row-" + rowIter);
				rowSelector.setAttribute("rowoffsetval", rowIter);
				
				bind(rowSelector, "click", function(e) {
					var el = (e.srcElement)?e.srcElement:e.target;
					 selectRow(parseInt(el.getAttribute("rowoffsetval"))	);

				row.appendChild(rowSelector);
				
			    for (var iter = 0; iter < defaultColumnCount; iter++) {
					var cell = getDefaultCell();
					cell.setAttribute("rowoffsetval", rowIter);
					cell.setAttribute("columnoffsetval", iter);
					row.appendChild(cell);
				}
			}

		};
		
		var availableWidth = getAvailableWidth();
		var availableHeight = getAvaialbleHeight();
		init();
